// Utilidad para resolver rutas de imágenes respetando contratos y el BASE_URL
function resolveImageSrc(rawImage) {
  if (!rawImage) return null;
  if (/^https?:\/\//i.test(rawImage)) return rawImage;
  // Si la ruta ya es absoluta desde la raíz, se respeta; sino, se anexa BASE_URL
  return rawImage.startsWith('/')
    ? rawImage
    : `${import.meta.env.BASE_URL || '/'}${rawImage}`;
}

// Componente seguro para manejo de imágenes con fallback
function ProductImage({ product }) {
  const [failed, setFailed] = useState(false);
  const src = resolveImageSrc(product.image_url || product.image);

  if (!src || failed) {
    return (
      <div className="w-full h-full flex items-center justify-center bg-slate-100">
        <Sun size={48} className="text-amber-500/40" />
      </div>
    );
  }

  return (
    <img
      src={src}
      alt={product.name}
      className="w-full h-full object-cover"
      loading="lazy"
      onError={() => setFailed(true)}
    />
  );
}
